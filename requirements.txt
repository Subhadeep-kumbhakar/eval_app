from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from api.database import engine, Base
from api.routers import auth, exams, submissions

# Automatically create all PostgreSQL tables
Base.metadata.create_all(bind=engine)

app = FastAPI(
    title="Student-Teacher Evaluation Platform API",
    version="2.0.0",
    description="Scalable RAG Exam Generation & Evaluation System",
)

app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

app.include_router(auth.router)
app.include_router(exams.router)
app.include_router(submissions.router)


@app.get("/")
def health_check():
    return {"status": "online", "system": "eval_app API"}
